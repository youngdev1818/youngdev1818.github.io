function makeRain(canvas, speed){ const ctx = canvas.getContext('2d'); let dpr = 1, cols = 0, step = FONT, drops = [], raf = null; function setup(){ dpr = Math.min(devicePixelRatio || 1, 2); // cap DPR for mobile perf canvas.width = Math.round(canvas.clientWidth * dpr); canvas.height = Math.round(canvas.clientHeight * dpr); step = FONT * dpr; ctx.font = `${step}px "Noto Sans Egyptian Hieroglyphs", monospace`; ctx.textBaseline = 'top'; cols = Math.ceil(canvas.width / step); drops = Array.from({length: cols}, () => Math.random() * -40); } function frame(){ ctx.fillStyle = 'rgba(0,0,0,0.09)'; // fade = the tail ctx.fillRect(0, 0, canvas.width, canvas.height); for (let i = 0; i < cols; i++){ const x = i * step, y = drops[i] * step; if (y > -step && y < canvas.height + step){ ctx.fillStyle = 'rgba(150,255,246,0.95)'; // bright head glyph ctx.fillText(GLYPHS[(Math.random() * GLYPHS.length) | 0], x, y); } drops[i] += speed; if (y > canvas.height && Math.random() > 0.975) drops[i] = Math.random() * -20; } raf = requestAnimationFrame(frame); } const start = () => { if (!raf && !reduce.matches){ setup(); frame(); } }; const stop = () => { if (raf){ cancelAnimationFrame(raf); raf = null; } }; addEventListener('resize', () => { stop(); start(); }); new IntersectionObserver(([e]) => e.isIntersecting ? start() : stop()).observe(canvas); document.addEventListener('visibilitychange', () => document.hidden ? stop() : start()); return { start, stop }; } makeRain(document.querySelector('.rain--center'), 0.45).start(); makeRain(document.querySelector('.rain--sides'), 0.75).start();
